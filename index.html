/** * Web API エントリポイント */ function doGet(e) { let result = { success: false, error: '不明なリクエスト' }; try { const params = e ? e.parameter : {}; const action = params.action || ''; if (action === 'product') { const barcode = params.barcode; const isReverse = params.reverse === 'true'; result = getProductDataFromShippingSheet(barcode, isReverse); } else if (action === 'recordCount') { const barcode = params.barcode; const storeCode = params.storeCode; const count = Number(params.count) || 1; result = markShippingCellAsGray(barcode, storeCode, count); } else if (action === 'recordBatch') { const barcode = params.barcode; const dataJson = params.data || '{}'; const storeCountsMap = JSON.parse(dataJson); result = markBatchShippingCellsAsGray(barcode, storeCountsMap); } else if (action === 'getProgressList') { result = { success: true, products: [] }; } else if (action === 'updateProgressSheet') { result = { success: true, message: "出荷データ入力シートを同期しました" }; } else { result = { success: false, error: '不正なアクションです: ' + action }; } } catch (err) { result = { success: false, error: 'GAS内部エラー: ' + err.toString() }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } /** * 【超高速化版】背景色チェックを廃止し、店舗の必要数のみを瞬時に取得する */ function getProductDataFromShippingSheet(barcode, isReverse) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('出荷データ入力'); if (!sheet) { return { success: false, error: '「出荷データ入力」シートが見つかりません' }; } const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow < 5 || lastCol < 4) { return { success: false, error: '「出荷データ入力」シートにデータが存在しません' }; } // ヘッダー情報（店舗コードや名前）を一括取得 const headerRange = sheet.getRange(1, 1, 4, lastCol).getValues(); const row3 = headerRange[2]; let storeStartCol = -1; for (let col = 0; col < row3.length; col++) { if (String(row3[col]).trim() !== '' && col >= 3) { storeStartCol = col; break; } } if (storeStartCol === -1) storeStartCol = 3; const numStoreCols = lastCol - storeStartCol; const storeNos = headerRange[0].slice(storeStartCol, lastCol); const storeCodes = headerRange[2].slice(storeStartCol, lastCol); const storeNames = headerRange[3].slice(storeStartCol, lastCol); const targetBarcode = String(barcode || '').trim(); // バーコード列（2列目）を一括取得して該当行を探す const barcodeValues = sheet.getRange(5, 2, lastRow - 4, 1).getValues(); let targetRowIndex = -1; for (let i = 0; i < barcodeValues.length; i++) { if (String(barcodeValues[i][0] || '').trim() === targetBarcode) { targetRowIndex = i + 5; break; } } if (targetRowIndex === -1) { return { success: false, error: '該当商品（' + targetBarcode + '）の出荷データが見つかりません' }; } // ★重い背景色取得(.getBackgrounds())をカットし、数値データ(.getValues())のみで爆速化 const targetRowValues = sheet.getRange(targetRowIndex, 1, 1, lastCol).getValues()[0]; const stores = []; let totalQty = 0; for (let c = 0; c < numStoreCols; c++) { const currentCol = storeStartCol + c; const qty = Number(targetRowValues[currentCol]) || 0; if (qty > 0) { const storeNo = String(storeNos[c] || '').trim(); const storeCode = String(storeCodes[c] || '').trim(); const storeName = String(storeNames[c] || '').trim(); if (storeCode !== '') { stores.push({ storeNo: storeNo, storeCode: storeCode, storeName: storeName, qty: qty, completedQty: 0, isCompleted: false }); totalQty += qty; } } } if (stores.length === 0) { return { success: false, error: '該当商品（' + targetBarcode + '）の店舗指示数がすべて0です' }; } if (isReverse) { stores.reverse(); } return { success: true, barcode: targetBarcode, itemCode: targetBarcode, totalQty: totalQty, stores: stores }; } /** * ピッキング完了時（個別店舗）：該当店舗セルを直接グレーアウト */ function markShippingCellAsGray(barcode, storeIdentifier, count) { const lock = LockService.getScriptLock(); if (!lock.tryLock(20000)) { return { success: false, error: '混雑しています。もう一度実行してください。' }; } try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('出荷データ入力'); if (!sheet) return { success: false, error: '「出荷データ入力」シートが見つかりません' }; const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); const targetBarcode = String(barcode || '').trim(); const targetId = String(storeIdentifier || '').trim(); const barcodeValues = sheet.getRange(5, 2, lastRow - 4, 1).getValues(); let targetRowIndex = -1; for (let i = 0; i < barcodeValues.length; i++) { if (String(barcodeValues[i][0] || '').trim() === targetBarcode) { targetRowIndex = i + 5; break; } } if (targetRowIndex === -1) { return { success: false, error: '商品が見つかりません: ' + targetBarcode }; } const row3 = sheet.getRange(3, 1, 1, lastCol).getValues()[0]; let storeStartCol = -1; for (let col = 0; col < row3.length; col++) { if (String(row3[col]).trim() !== '' && col >= 3) { storeStartCol = col; break; } } if (storeStartCol === -1) storeStartCol = 3; const numStoreCols = lastCol - storeStartCol; const headerRange = sheet.getRange(1, storeStartCol + 1, 4, numStoreCols).getValues(); const storeNos = headerRange[0]; const storeCodes = headerRange[2]; const storeNames = headerRange[3]; let targetColIndex = -1; for (let c = 0; c < numStoreCols; c++) { const sNo = String(storeNos[c] || '').trim(); const sCode = String(storeCodes[c] || '').trim(); const sName = String(storeNames[c] || '').trim(); if (targetId === sCode || targetId === sNo || targetId === sName) { targetColIndex = storeStartCol + 1 + c; break; } } if (targetColIndex === -1) { return { success: false, error: '店舗が見つかりません: ' + targetId }; } const cell = sheet.getRange(targetRowIndex, targetColIndex); cell.setBackground('#d1d5db'); cell.setFontColor('#4b5563'); return { success: true }; } catch (err) { return { success: false, error: 'グレー反映エラー: ' + err.toString() }; } finally { lock.releaseLock(); } } /** * 全店舗ピッキング完了時の一括グレーアウト（完了時にしっかりグレーに塗る処理） */ function markBatchShippingCellsAsGray(barcode, storeCountsMap) { const lock = LockService.getScriptLock(); if (!lock.tryLock(20000)) { return { success: false, error: '混雑しています。もう一度実行してください。' }; } try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('出荷データ入力'); if (!sheet) return { success: false, error: '「出荷データ入力」シートが見つかりません' }; const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); const targetBarcode = String(barcode || '').trim(); const barcodeValues = sheet.getRange(5, 2, lastRow - 4, 1).getValues(); let targetRowIndex = -1; for (let i = 0; i < barcodeValues.length; i++) { if (String(barcodeValues[i][0] || '').trim() === targetBarcode) { targetRowIndex = i + 5; break; } } if (targetRowIndex === -1) { return { success: false, error: '商品が見つかりません: ' + targetBarcode }; } const row3 = sheet.getRange(3, 1, 1, lastCol).getValues()[0]; let storeStartCol = -1; for (let col = 0; col < row3.length; col++) { if (String(row3[col]).trim() !== '' && col >= 3) { storeStartCol = col; break; } } if (storeStartCol === -1) storeStartCol = 3; const numStoreCols = lastCol - storeStartCol; const storeCodes = sheet.getRange(3, storeStartCol + 1, 1, numStoreCols).getValues()[0]; for (let c = 0; c < numStoreCols; c++) { const sCode = String(storeCodes[c] || '').trim(); if (storeCountsMap.hasOwnProperty(sCode)) { const targetColIndex = storeStartCol + 1 + c; const cell = sheet.getRange(targetRowIndex, targetColIndex); cell.setBackground('#d1d5db'); cell.setFontColor('#4b5563'); } } sheet.getRange(targetRowIndex, 1).setBackground('#d1d5db'); sheet.getRange(targetRowIndex, 1).setFontColor('#4b5563'); return { success: true }; } catch (err) { return { success: false, error: '一括反映エラー: ' + err.toString() }; } finally { lock.releaseLock(); } }
