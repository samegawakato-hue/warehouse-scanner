// ★取得したGASのウェブアプリURLをここに貼り付け const GAS_URL = "https://script.google.com/macros/s/ここに取得したIDを入れる/exec"; function fetchProductData(barcode) { showStatus('商品データ取得中...', 'success'); const url = `${GAS_URL}?action=product&barcode=${encodeURIComponent(barcode)}`; fetch(url) .then(response => response.json()) .then(data => { if (data.success) { currentData = data; storeCounts = {}; data.stores.forEach(s => storeCounts[s.storeCode] = 0); renderProductData(data); switchMode('OUT_STORE'); showStatus('商品確認完了。店舗箱のバーコードをスキャンしてください', 'success'); } else { triggerError(data.error || '未登録の商品です'); } }) .catch(err => { console.error(err); triggerError('通信エラー：データの取得に失敗しました'); }); }
