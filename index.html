function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var result = {}; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("出荷データ入力"); if (!sheet) { throw new Error("「出荷データ入力」シートが見つかりません"); } var rows = sheet.getDataRange().getValues(); if (action === "product") { var barcode = e.parameter.barcode; var isReverse = e.parameter.reverse === "true"; var stores = []; var totalQty = 0; // 2行目（インデックス1）からデータ走査 for (var i = 1; i < rows.length; i++) { // B列（インデックス1）がバーコード if (String(rows[i][1]).trim() === String(barcode).trim()) { var storeNo = rows[i][0]; // A列: 箱No. var storeCode = rows[i][2]; // C列: 店舗コード var qty = Number(rows[i][3]) || 1; // D列: 必要数 var status = rows[i][4]; // E列: 状態 var storeName = rows[i][5] || ("店舗" + storeNo); // F列: 店舗名 var completedQty = (status === "完了") ? qty : 0; stores.push({ storeCode: storeCode, storeNo: storeNo, storeName: storeName, qty: qty, completedQty: completedQty }); totalQty += qty; } } if (stores.length > 0) { if (isReverse) stores.reverse(); result = { success: true, barcode: barcode, totalQty: totalQty, stores: stores }; } else { result = { success: false, error: "該当するバーコードが見つかりません" }; } } else if (action === "recordCount") { var barcode = e.parameter.barcode; var storeCode = e.parameter.storeCode; var totalCount = 0; var completedCount = 0; for (var i = 1; i < rows.length; i++) { // B列がバーコード、C列が店舗コードの行を特定 if (String(rows[i][1]).trim() === String(barcode).trim()) { totalCount++; if (String(rows[i][2]).trim() === String(storeCode).trim()) { // E列（5列目）に「完了」を書き込む sheet.getRange(i + 1, 5).setValue("完了"); // 該当行全体をグレーアウト sheet.getRange(i + 1, 1, 1, sheet.getLastColumn()).setBackground("#d9d9d9"); rows[i][4] = "完了"; // 判定用メモリも更新 } // すでに完了しているものをカウント if (rows[i][4] === "完了") { completedCount++; } } } // この商品の全店舗へのピッキングが完了していたら、A1の型番も自動でグレーアウト if (totalCount > 0 && completedCount >= totalCount) { sheet.getRange("A1").setBackground("#d9d9d9"); } result = { success: true, completedCount: completedCount, totalCount: totalCount }; } } catch (err) { result = { success: false, error: err.toString() }; } // アプリ側（JSONP）へ返す処理 var jsonString = callback + "(" + JSON.stringify(result) + ")"; return ContentService.createTextOutput(jsonString).setMimeType(ContentService.MimeType.JAVASCRIPT); }
